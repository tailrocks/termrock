## Usage

```rust
use termrock::{style::DesignSystem, widgets::{BarDatum, BarSeries}};

let system = DesignSystem::default();
let bars = [BarDatum { label: "cpu", fraction: 0.72 }];
let series = BarSeries::new(&bars, &system);
```
