## Usage

```rust
use termrock::{Theme, style::ColorCapability, widgets::{DesignInspector, DesignInspectorFrame}};

let theme = Theme::default();
let frame = DesignInspectorFrame {
    focused: Some("list"),
    layer: Some("root"),
    capability: ColorCapability::Truecolor,
    density: "comfortable",
};
let _inspector = DesignInspector::new(frame, &theme);
```
