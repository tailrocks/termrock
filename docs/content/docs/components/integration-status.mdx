## Usage

```rust
use termrock::widgets::{
    example_integrations, IntegrationStatus, IntegrationStatusOutcome,
    IntegrationStatusState,
};

let mut state = IntegrationStatusState::new();
state.set_entries(example_integrations());
IntegrationStatus::new(&system).paint(area, buf, &mut state);

match state.handle_key(key) {
    IntegrationStatusOutcome::PermissionRequested { id, .. } => { let _ = id; }
    IntegrationStatusOutcome::RestartRequested { id } => { let _ = id; }
    _ => {}
}
```
