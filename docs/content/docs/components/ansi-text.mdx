## Usage

```rust
use termrock::widgets::{
    AnsiStream, AnsiText, AnsiTextState, AnsiParseOptions, lines_for_log, parse_lines,
};

// Streaming tool / PTY output
let mut stream = AnsiStream::new(AnsiParseOptions::for_system(&system))
    .with_max_lines(10_000);
stream.feed(bytes);
stream.drain_into_lines(|line| log_pane.append(line));

// Paint pre-parsed lines
let lines = parse_lines(src, &AnsiParseOptions::for_system(&system));
let mut state = AnsiTextState::new();
AnsiText::lines(&lines, &system).paint(area, buf, &mut state);
```
