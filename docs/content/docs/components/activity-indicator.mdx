## Usage

The Code view above and this source excerpt use the exact paint setup registered for `activity-indicator/basic`. Interaction state is long-lived in the catalog interactor; the host only forwards backend-neutral events.

```rust
use ratatui::prelude::*;
use termrock::{input::*, interaction::*, runtime::*, style::*, widgets::*};

// Exact paint setup used by the shared catalog story.
fn activity_indicator_story(frame: &mut Frame<'_>, area: Rect, system: &DesignSystem) {
    use std::time::Duration;
    use termrock::runtime::FrameTick;
    use termrock::runtime::Instant;
    use termrock::style::MotionPolicy;
    let mut state = SpinnerState::new();
    state.set_phase(ActivityPhase::Reconnecting);
    let tick = crate::demo::demo_tick(200);
    ActivityIndicator::new("Reconnecting to agent", system)
        .detail("attempt 2/5 · backoff 1.2s")
        .paint(area, frame.buffer_mut(), &state, tick, MotionPolicy::Full);
}
```
