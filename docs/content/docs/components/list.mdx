## Usage

```rust
use ratatui_core::text::Line;
use termrock::{
    style::DesignSystem,
    widgets::{List, ListClickPolicy, ListRow, ListState},
};

let system = DesignSystem::phosphor();
let rows = [
    ListRow::item("alpha", Line::from("Alpha")).badge(Line::from("12 ms")),
    ListRow::item("beta", Line::from("Beta")).secondary(Line::from("ready")),
];
let list = List::from_system(&rows, &system);
let mut state = ListState::new(Some("alpha"));
state.set_click_policy(ListClickPolicy::Select); // click selects; Enter activates
let outcome = state.select_next(&rows);
```

Empty lists paint an optional message:

```rust
List::from_system(&[], &system).empty_message(Line::from("No results"));
```
