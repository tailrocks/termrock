## Usage

```rust
use termrock::{Theme, widgets::{Timeline, TimelineEvent}};

let theme = Theme::default();
let events = [TimelineEvent { when: "12:01", text: "Started", active: true }];
let timeline = Timeline::new(&events, &theme);
```
