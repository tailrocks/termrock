## Usage

```rust
use termrock::{style::DesignSystem, widgets::{ColumnModel, DataColumn, DataColumnWidth, DataTable, DataTableState}};

let system = DesignSystem::default();
let columns = ColumnModel::new(vec![DataColumn::new("c", "C", DataColumnWidth::Min(4))]);
let rows: [(u64, &[&str]); 0] = [];
```
