## Usage

```rust
use termrock::style::{DesignSystem, Motion};
use termrock::widgets::{Tooltip, TooltipContent, TooltipState};

let mut state = TooltipState::new();
state.set_pointer_over(hovering);
state.set_focus_within(anchor_focused);
let _ = state.advance(tick, motion); // Motion::Reduced → no delay

Tooltip::content(
    TooltipContent::plain("Icon meaning").essential_elsewhere(true),
    &system,
)
.paint(placed_rect, buf, &state);
```
