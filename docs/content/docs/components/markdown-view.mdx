## Usage

```rust
use termrock::{Theme, widgets::{MarkdownBlock, MarkdownBlockKind, MarkdownView}};

let theme = Theme::default();
let blocks = [MarkdownBlock { kind: MarkdownBlockKind::Heading, text: "Plan" }];
let view = MarkdownView::new(&blocks, &theme);
```
