## Usage

The Code view above and this source excerpt use the exact paint setup registered for `metric-radar/basic`. Interaction state is long-lived in the catalog interactor; the host only forwards backend-neutral events.

```rust
use ratatui::prelude::*;
use termrock::{input::*, interaction::*, runtime::*, style::*, widgets::*};

// Exact paint setup used by the shared catalog story.
fn metric_radar_basic(frame: &mut Frame<'_>, area: Rect, system: &DesignSystem) {
    use termrock::widgets::{MetricAxis, MetricRadar, MetricSeries};
    let axes = [
        MetricAxis::new("cpu"),
        MetricAxis::new("mem"),
        MetricAxis::new("io"),
        MetricAxis::new("net"),
    ];
    let a = [0.8, 0.5, 0.3, 0.6];
    let b = [0.4, 0.7, 0.5, 0.2];
    let series = [
        MetricSeries::new("svc-a", &a),
        MetricSeries::new("svc-b", &b),
    ];
    frame.render_widget(
        MetricRadar::new(&axes, &series, system).title("services"),
        area,
    );
}
```
