## Usage

```rust
use termrock::widgets::{Kbd, format_chord, ChordFormat, Platform, ModifierStyle};
use termrock::keymap::KeyChord;
use termrock::input::KeyCode;

Kbd::from_chord(KeyChord::ctrl(KeyCode::Char('s')), &system)
    .keycap()
    .paint(area, buf);

// From active keymap
Kbd::for_action(&map, Action::Save, &system);

// Platform-aware
format_chord(
    KeyChord::ctrl(KeyCode::Char('c')),
    ChordFormat::docs()
        .platform(Platform::Mac)
        .modifiers(ModifierStyle::Symbols),
);
```
