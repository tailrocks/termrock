## Usage

```rust
use termrock::{Theme, widgets::{StreamItem, StreamItemKind, StreamView}};

let theme = Theme::default();
let items = [StreamItem { id: "u1", kind: StreamItemKind::User, text: "Hello", folded: false }];
let stream = StreamView::new(&items, &theme);
```
