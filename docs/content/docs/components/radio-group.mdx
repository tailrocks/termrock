## Usage

```rust
use termrock::widgets::{
    RadioGroup, RadioOption, RadioState, RadioSelectionPolicy,
};

let options = [
    RadioOption::new("plan", "Plan").description("Read-only"),
    RadioOption::new("build", "Build").badge("default"),
    RadioOption::new("ask", "Ask"),
];
let mut state = RadioState::new(Some("build"));
state.set_surface_focused(true);

RadioGroup::new(&options, &system)
    .legend("Workbench mode")
    .paint(area, buf, &mut state);

// Destructive / review: move without commit
state.set_policy(RadioSelectionPolicy::ActivateToSelect);
```
